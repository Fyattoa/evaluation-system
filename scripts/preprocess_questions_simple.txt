import os
import re
from tkinter import filedialog, Tk


def add_question_prefix(input_path, output_path):
    with open(input_path, 'r', encoding='utf-8') as f:
        lines = f.readlines()

    new_lines = []
    for line in lines:
        # 匹配以纯数字 + 点号 开头的行（数字后没有减号或数字）
        # 匹配模式：行首可选空格，然后数字，点号，断言后面不是减号（避免匹配到 1-1 之类）
        if re.match(r'^\s*\d+\.(?!-)', line):
            # 在行首添加“问题”二字（保留原有缩进）
            new_lines.append("问题" + line)
        else:
            new_lines.append(line)

    with open(output_path, 'w', encoding='utf-8') as f:
        f.writelines(new_lines)
    print(f"处理完成！已保存至：{output_path}")


def main():
    root = Tk()
    root.withdraw()

    # 选择输入文件
    input_file = filedialog.askopenfilename(
        title="请选择要处理的文本文件",
        filetypes=[("Text files", "*.txt"), ("All files", "*.*")]
    )
    if not input_file:
        print("未选择文件，退出。")
        return

    # 选择输出文件（默认桌面，用户可自定义文件名）
    desktop = os.path.join(os.path.expanduser("~"), "Desktop")
    output_file = filedialog.asksaveasfilename(
        title="保存处理后的文件",
        initialdir=desktop,
        defaultextension=".txt",
        filetypes=[("Text files", "*.txt"), ("All files", "*.*")]
    )
    if not output_file:
        print("未选择保存路径，退出。")
        return

    add_question_prefix(input_file, output_file)
    root.destroy()


if __name__ == "__main__":
    main()
